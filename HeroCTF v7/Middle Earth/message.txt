import math

# ------------- Geometry helpers -------------

def distance(p1, p2):
    return math.hypot(p1[0] - p2[0], p1[1] - p2[1])


def angle_to_target(ship_pos, target_pos):
    """
    Convert target vector into game angle:
    0° = North, 90° = East, 180° = South, 270° = West
    """
    sx, sy = ship_pos
    tx, ty = target_pos
    dx = tx - sx
    dy = ty - sy

    if dx == 0 and dy == 0:
        return 0

    # Game direction: vector(angle) = (sin(angle), -cos(angle))
    ang_rad = math.atan2(dx, -dy)
    ang_deg = (math.degrees(ang_rad) + 360) % 360
    return ang_deg


def normalize_angle_diff(a, b):
    # minimal signed difference a-b in degrees, in [-180, 180]
    return (a - b + 540) % 360 - 180


# ------------- Island logic -------------

def island_validation_point(island, sea_size):
    """
    Point where we expect to validate:
      Type 1 (East)  -> just east  of island
      Type 2 (South) -> just south of island
      Type 3 (West)  -> just west  of island
      Type 4 (North) -> just north of island
    """
    ix = island["position"]["x"]
    iy = island["position"]["y"]
    r = island["radius"]
    t = island["type"]

    margin = 5
    border_margin = 5
    offset = r + margin  # a bit outside island radius

    if t == 1:  # East => x > ix
        x = min(ix + offset, sea_size - border_margin)
        y = iy
    elif t == 3:  # West => x < ix
        x = max(ix - offset, border_margin)
        y = iy
    elif t == 2:  # South => y > iy
        x = ix
        y = min(iy + offset, sea_size - border_margin)
    else:  # t == 4, North => y < iy
        x = ix
        y = max(iy - offset, border_margin)

    return (x, y)


def choose_next_island(ship_pos, islands, sea_size):
    """
    Among unvalidated islands, choose the one whose validation point
    is closest to our current position.
    """
    best_idx = None
    best_dist = float("inf")

    for idx, island in enumerate(islands):
        if island.get("validated"):
            continue

        vp = island_validation_point(island, sea_size)
        d = distance(ship_pos, vp)

        if d < best_dist:
            best_dist = d
            best_idx = idx

    return best_idx, best_dist


# ------------- Barrel & memory logic -------------

def is_banned_barrel(x, y, banned, ban_dist=20.0):
    for bx, by in banned:
        if distance((x, y), (bx, by)) <= ban_dist:
            return True
    return False


def choose_next_barrel(ship_pos, barrels, banned):
    """
    Choose the closest uncollected barrel that is not banned.
    """
    best_idx = None
    best_dist = float("inf")

    for idx, barrel in enumerate(barrels):
        if barrel.get("collected"):
            continue

        bx = barrel["position"]["x"]
        by = barrel["position"]["y"]

        if is_banned_barrel(bx, by, banned):
            continue

        d = distance(ship_pos, (bx, by))

        if d < best_dist:
            best_dist = d
            best_idx = idx

    return best_idx, best_dist


# data format (max 64 chars):
# "Bx1,y1;x2,y2|Ppx,py,c"
#   B: banned barrels (up to 3) as "x,y" pairs separated by ';'
#   P: previous barrel target (px,py) and close_turns c
#
# Example: "B100,200;300,400|P500,600,3"

def parse_data(data_str):
    banned = []
    prev = None  # (px, py, close_turns)

    if not data_str:
        return banned, prev

    try:
        parts = data_str.split('|')
        for part in parts:
            if not part:
                continue
            if part[0] == 'B':
                rest = part[1:]
                if not rest:
                    continue
                tokens = rest.split(';')
                for t in tokens:
                    if not t:
                        continue
                    xs = t.split(',')
                    if len(xs) != 2:
                        continue
                    x = int(xs[0])
                    y = int(xs[1])
                    banned.append((x, y))
            elif part[0] == 'P':
                rest = part[1:]
                if not rest:
                    continue
                xs = rest.split(',')
                if len(xs) != 3:
                    continue
                px = int(xs[0])
                py = int(xs[1])
                c = int(xs[2])
                prev = (px, py, c)
    except Exception:
        # If parsing fails, just reset memory
        banned = []
        prev = None

    return banned, prev


def encode_data(banned, prev):
    # Keep at most 3 banned barrels to stay well under 64 chars
    banned = banned[-3:]

    parts = []

    if banned:
        b_tokens = []
        for (x, y) in banned:
            b_tokens.append(f"{int(x)},{int(y)}")
        parts.append("B" + ";".join(b_tokens))

    if prev is not None:
        px, py, c = prev
        parts.append(f"P{int(px)},{int(py)},{int(c)}")

    s = "|".join(parts)

    # Safety check: if somehow too long, drop prev, then banned
    if len(s) > 64:
        if banned:
            b_tokens = [f"{int(x)},{int(y)}" for (x, y) in banned[-3:]]
            s2 = "B" + ";".join(b_tokens)
            if len(s2) <= 64:
                return s2
        return ""

    return s


# ------------- Main bot -------------

def make_move(state):
    sea_size = state.get("sea_size", 1000)

    ship = state["your_ship"]
    sx = ship["position"]["x"]
    sy = ship["position"]["y"]
    ship_pos = (sx, sy)

    vx = ship["velocity"]["x"]
    vy = ship["velocity"]["y"]
    speed = math.hypot(vx, vy)

    ship_angle = ship.get("angle", 0)

    islands = state.get("islands", [])
    barrels = state.get("barrels", [])

    # Load compact memory
    banned, prev = parse_data(state.get("data") or "")

    # Default target
    target_pos = (sx, sy - 1)
    target_kind = "none"
    target_barrel_idx = None

    # Choose target: islands first, but detour for close barrels
    island_idx, island_dist = choose_next_island(ship_pos, islands, sea_size)
    barrel_idx, barrel_dist = choose_next_barrel(ship_pos, barrels, banned)

    if island_idx is not None:
        # Prefer islands; only detour to barrel if it's much closer
        take_barrel = (
            barrel_idx is not None
            and barrel_dist < 0.7 * island_dist
            and barrel_dist < 150
        )

        if take_barrel:
            b = barrels[barrel_idx]
            target_pos = (b["position"]["x"], b["position"]["y"])
            target_kind = "barrel"
            target_barrel_idx = barrel_idx
        else:
            island = islands[island_idx]
            target_pos = island_validation_point(island, sea_size)
            target_kind = "island"
    elif barrel_idx is not None:
        # No islands left -> farm barrels
        b = barrels[barrel_idx]
        target_pos = (b["position"]["x"], b["position"]["y"])
        target_kind = "barrel"
        target_barrel_idx = barrel_idx

    # Aim and throttle
    angle = angle_to_target(ship_pos, target_pos)
    dist = distance(ship_pos, target_pos)

    if dist > 250:
        acceleration = 100
    elif dist > 120:
        acceleration = 90
    elif dist > 50:
        acceleration = 60
    else:
        # Close to target: slow down a bit if we're moving fast
        if speed > 12:
            acceleration = -30
        else:
            acceleration = 30

    # If we're facing far away from the desired direction, don't over-thrust
    angle_diff = abs(normalize_angle_diff(angle, ship_angle))
    if angle_diff > 100:
        acceleration = min(acceleration, 40)

    acceleration = max(-100, min(100, acceleration))

    # ----- Barrel “frustration” logic with compact memory -----
    CLOSE_DIST = 25.0
    FRUSTRATION_LIMIT = 8

    if target_kind == "barrel" and target_barrel_idx is not None:
        bx, by = target_pos

        # Previous barrel target?
        same_barrel = False
        close_turns = 0
        if prev is not None:
            px, py, prev_c = prev
            if distance((bx, by), (px, py)) < 5.0:
                same_barrel = True
                close_turns = prev_c

        # Update close_turns if we're close
        if dist < CLOSE_DIST:
            if same_barrel:
                close_turns += 1
            else:
                close_turns = 1
        else:
            # If far from barrel, reset counter if it was the same barrel
            if same_barrel:
                close_turns = 0

        prev = (bx, by, close_turns)

        barrel_obj = barrels[target_barrel_idx]
        # If we've been very close for many turns and still haven't collected it,
        # assume it's effectively unreachable (or already looted by opponent)
        if not barrel_obj.get("collected", False) and close_turns >= FRUSTRATION_LIMIT:
            banned.append((bx, by))
            prev = None  # stop tracking this barrel
    else:
        # Not currently chasing a barrel; reset prev if it was barrel-related
        if prev is not None:
            prev = None

    # Save memory back
    data_str = encode_data(banned, prev)

    return {
        "acceleration": acceleration,
        "angle": int(angle) % 360,
        "data": data_str,
    }
